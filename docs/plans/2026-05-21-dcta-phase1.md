"""Synthetic AnnData fixtures for DCTA tests."""
import numpy as np
import pandas as pd
import anndata as ad
import pytest


@pytest.fixture
def synthetic_adata():
    """3-lineage star graph: HSC -> {Mye, Ery, Lym}, 500 cells, 100 genes.

    Cells per group:
      HSC: 125, Mye: 125, Ery: 125, Lym: 125
    Genes:
      - 8 HSC markers (CRHBP, AVP, HLF, MEIS1, MLLT3, HOPX, HEMGN, PROM1)
      - 10 per-lineage markers (MYE_1..10, ERY_1..10, LYM_1..10)
      - 62 noise genes (NOISE_1..62)
    Counts are negative-binomial-like ints. Lineage-marker expression is
    enriched in the corresponding lineage; HSC markers are enriched in HSC.
    Returns a raw-count AnnData (no normalization done).
    """
    rng = np.random.default_rng(seed=0)
    n_per_group = 125
    n_obs = 4 * n_per_group  # 500
    lineages = (["HSC"] * n_per_group + ["Mye"] * n_per_group +
                ["Ery"] * n_per_group + ["Lym"] * n_per_group)

    hsc_markers = ["CRHBP", "AVP", "HLF", "MEIS1", "MLLT3", "HOPX", "HEMGN", "PROM1"]
    mye_markers = [f"MYE_{i}" for i in range(1, 11)]
    ery_markers = [f"ERY_{i}" for i in range(1, 11)]
    lym_markers = [f"LYM_{i}" for i in range(1, 11)]
    noise_genes = [f"NOISE_{i}" for i in range(1, 63)]
    var_names = hsc_markers + mye_markers + ery_markers + lym_markers + noise_genes
    n_vars = len(var_names)

    # Baseline low expression for all
    base = rng.negative_binomial(n=2, p=0.4, size=(n_obs, n_vars)).astype(np.float32)

    # Enrichment helper
    def enrich(group_label, gene_list, scale=20):
        gidx = [var_names.index(g) for g in gene_list]
        cmask = np.array([l == group_label for l in lineages])
        base[np.ix_(cmask, gidx)] += rng.negative_binomial(n=8, p=0.3,
                                                           size=(cmask.sum(), len(gidx))) * (scale / 10)

    enrich("HSC", hsc_markers, scale=25)
    enrich("Mye", mye_markers, scale=25)
    enrich("Ery", ery_markers, scale=25)
    enrich("Lym", lym_markers, scale=25)

    # Add a couple of mitochondrial-prefix genes (small, low expression)
    mt_names = ["MT-CO1", "MT-CO2", "MT-ND1"]
    mt_counts = rng.negative_binomial(n=1, p=0.5, size=(n_obs, len(mt_names))).astype(np.float32)
    X = np.concatenate([base, mt_counts], axis=1)
    var_names = var_names + mt_names

    obs = pd.DataFrame({"true_lineage": lineages,
                        "cell_id": [f"cell_{i:04d}" for i in range(n_obs)]})
    obs = obs.set_index("cell_id")
    var = pd.DataFrame(index=pd.Index(var_names, name="gene"))

    adata = ad.AnnData(X=X, obs=obs, var=var)
    return adata


@pytest.fixture
def synthetic_adata_disconnected(synthetic_adata):

"""I/O helpers for DCTA."""
from __future__ import annotations
import os
import anndata as ad
from dcta import DCTADataError


def load_h5ad(path: str) -> ad.AnnData:
    """Load an .h5ad file from disk.

    Parameters
    ----------
    path : str
        Path to .h5ad file. Must exist.

    Returns
    -------
    AnnData
        The loaded annotated data matrix.

    Raises
    ------
    DCTADataError
        If the file does not exist or cannot be read as h5ad.
    """
    if not os.path.exists(path):
        raise DCTADataError(f"AnnData file not found: {path}")
    try:
        adata = ad.read_h5ad(path)
    except Exception as e:  # noqa: BLE001
        raise DCTADataError(f"Failed to read h5ad {path}: {e}") from e
    return adata
```

"""DCTA preprocessing: QC, normalization, HVG, optional cell-cycle regress-out."""
from __future__ import annotations
import numpy as np
import scanpy as sc
import anndata as ad
from dcta import DCTADataError
from dcta import defaults as D


# Tirosh et al. 2016 cell cycle gene lists (subset, human symbols).
# Source: regev_lab_cell_cycle_genes.txt (commonly distributed with scanpy).
_S_GENES = [
    "MCM5", "PCNA", "TYMS", "FEN1", "MCM2", "MCM4", "RRM1", "UNG", "GINS2",
    "MCM6", "CDCA7", "DTL", "PRIM1", "UHRF1", "MLF1IP", "HELLS", "RFC2",
    "RPA2", "NASP", "RAD51AP1", "GMNN", "WDR76", "SLBP", "CCNE2", "UBR7",
    "POLD3", "MSH2", "ATAD2", "RAD51", "RRM2", "CDC45", "CDC6", "EXO1",
    "TIPIN", "DSCC1", "BLM", "CASP8AP2", "USP1", "CLSPN", "POLA1", "CHAF1B",
    "BRIP1", "E2F8",
]
_G2M_GENES = [
    "HMGB2", "CDK1", "NUSAP1", "UBE2C", "BIRC5", "TPX2", "TOP2A", "NDC80",
    "CKS2", "NUF2", "CKS1B", "MKI67", "TMPO", "CENPF", "TACC3", "FAM64A",
    "SMC4", "CCNB2", "CKAP2L", "CKAP2", "AURKB", "BUB1", "KIF11", "ANP32E",
    "TUBB4B", "GTSE1", "KIF20B", "HJURP", "CDCA3", "HN1", "CDC20", "TTK",
    "CDC25C", "KIF2C", "RANGAP1", "NCAPD2", "DLGAP5", "CDCA2", "CDCA8",
    "ECT2", "KIF23", "HMMR", "AURKA", "PSRC1", "ANLN", "LBR", "CKAP5",
    "CENPE", "CTCF", "NEK2", "G2E3", "GAS2L3", "CBX5", "CENPA",
]


def _validate(adata: ad.AnnData) -> None:
    if adata.n_obs == 0 or adata.n_vars == 0:
        raise DCTADataError("AnnData is empty (0 cells or 0 genes)")
    X = adata.X
    Xmin = X.min() if not hasattr(X, "toarray") else X.min()
    if Xmin < 0:
        raise DCTADataError(f"AnnData.X contains negative values (min={Xmin})")


def run(
    adata: ad.AnnData,
    *,
    copy: bool = False,
    regress_cell_cycle: bool = True,
    min_genes: int = D.MIN_GENES_PER_CELL,
    min_cells: int = D.MIN_CELLS_PER_GENE,
    pct_mito_max: float = D.PCT_MITO_MAX,
    n_top_genes: int = D.N_TOP_GENES,
) -> ad.AnnData:
    """Preprocess raw count AnnData for DCTA.

    Steps:
      1. Validate (no empty matrix, no negative values).
      2. QC filter (min_genes, min_cells, pct_mito).
      3. Save raw counts to .raw.
      4. Normalize total + log1p.
      5. Highly-variable genes.
      6. Optional cell-cycle scoring + regress-out.

    Returns the same AnnData (mutated) when copy=False.
    """
    a = adata.copy() if copy else adata
    _validate(a)

    # Mitochondrial percentage
    a.var["mt"] = a.var_names.str.startswith(("MT-", "mt-"))
    sc.pp.calculate_qc_metrics(a, qc_vars=["mt"], percent_top=None,
                               log1p=False, inplace=True)
    a.obs["n_counts"] = a.obs["total_counts"]
    a.obs["pct_mito"] = a.obs["pct_counts_mt"]

    # QC filter
    sc.pp.filter_cells(a, min_genes=min_genes)
    sc.pp.filter_genes(a, min_cells=min_cells)
    a = a[a.obs["pct_mito"] <= pct_mito_max, :].copy()

    # Save raw counts
    a.raw = a

    # Normalize + log
    sc.pp.normalize_total(a, target_sum=1e4)
    sc.pp.log1p(a)

    # HVG
    sc.pp.highly_variable_genes(a, n_top_genes=n_top_genes, flavor="seurat")

    # Cell cycle
    s_present = [g for g in _S_GENES if g in a.var_names]
    g2m_present = [g for g in _G2M_GENES if g in a.var_names]
    if s_present and g2m_present:
        sc.tl.score_genes_cell_cycle(a, s_genes=s_present, g2m_genes=g2m_present)
    else:
        # No cell-cycle genes available — write NaN columns so downstream code
        # can rely on the columns existing
        a.obs["S_score"] = np.nan
        a.obs["G2M_score"] = np.nan
        a.obs["phase"] = "unknown"

    if regress_cell_cycle and s_present and g2m_present:
        sc.pp.regress_out(a, ["S_score", "G2M_score"])
        sc.pp.scale(a, max_value=10)

    # Mirror result into original adata if copy=False and slicing happened
    return a
```

```python
"""DCTA diffusion stage: PCA, KNN, diffusion map with eigval-gap auto-n_dcs."""
from __future__ import annotations
import numpy as np
import scanpy as sc
import anndata as ad
from dcta import defaults as D


def _select_n_dcs_by_gap(eigvals: np.ndarray, min_dcs: int = 2,
                         max_dcs: int | None = None) -> int:
    """Pick n_dcs at the largest relative drop (eigenvalue gap).

    Skips index 0 (trivial eigenvalue ~1.0 of the diffusion operator).
    """
    e = np.asarray(eigvals, dtype=float)
    if max_dcs is None:
        max_dcs = len(e) - 1
    max_dcs = min(max_dcs, len(e) - 1)
    if max_dcs <= min_dcs:
        return max_dcs
    # Consider gaps from min_dcs..max_dcs (between e[i] and e[i+1])
    gaps = e[min_dcs:max_dcs] - e[min_dcs + 1:max_dcs + 1]
    rel_gaps = gaps / np.maximum(np.abs(e[min_dcs:max_dcs]), 1e-9)
    best = int(np.argmax(rel_gaps)) + min_dcs + 1
    return max(min_dcs, min(best, max_dcs))


def run(
    adata: ad.AnnData,
    *,
    copy: bool = False,
    n_comps_pca: int = D.N_COMPS_PCA,
    n_comps: int = D.N_COMPS_DIFFMAP,
    knn_k: int = D.KNN_PCA_K,
    random_state: int = D.RANDOM_STATE,
) -> ad.AnnData:
    """Run PCA → KNN → diffusion map, then pick n_dcs from eigenvalue gap.

    Mutates adata (or copy) and returns it.
    """
    a = adata.copy() if copy else adata

    sc.pp.pca(a, n_comps=min(n_comps_pca, a.n_vars - 1, a.n_obs - 1),
              random_state=random_state)
    sc.pp.neighbors(a, n_neighbors=knn_k, n_pcs=a.obsm["X_pca"].shape[1],
                    random_state=random_state)
    sc.tl.diffmap(a, n_comps=min(n_comps, a.n_obs - 2))

    eigvals = np.asarray(a.uns["diffmap_evals"])
    n_dcs = _select_n_dcs_by_gap(eigvals, min_dcs=2, max_dcs=len(eigvals) - 1)

    uns = a.uns.setdefault("dcta", {})
    uns["diffmap_eigvals"] = eigvals.tolist()
    uns["n_dcs"] = int(n_dcs)
    return a
```

```python
"""DCTA connectivity stage: diffusion-based KNN -> Leiden -> PAGA."""
from __future__ import annotations
import numpy as np
import scanpy as sc
import anndata as ad
from dcta import defaults as D


def run(
    adata: ad.AnnData,
    *,
    copy: bool = False,
    knn_k: int = D.KNN_DIFFMAP_K,
    resolution: float = D.LEIDEN_RESOLUTION,
    random_state: int = D.RANDOM_STATE,
) -> ad.AnnData:
    """Build diffusion-based KNN, run Leiden clustering and PAGA.

    Stores connectivities into obsp['dcta_knn'], cluster labels into
    obs['dcta_leiden'], and PAGA outputs into uns['dcta']['paga'].
    """
    a = adata.copy() if copy else adata

    n_dcs = int(a.uns.get("dcta", {}).get("n_dcs", a.obsm["X_diffmap"].shape[1] - 1))
    # Build neighbors on diffusion components
    sc.pp.neighbors(a, n_neighbors=knn_k, use_rep="X_diffmap",
                    n_pcs=n_dcs, random_state=random_state, key_added="dcta")
    # Move the resulting connectivities into obsp['dcta_knn'] for clarity
    a.obsp["dcta_knn"] = a.obsp["dcta_connectivities"]

    sc.tl.leiden(a, neighbors_key="dcta", resolution=resolution,
                 random_state=random_state, key_added="dcta_leiden")

    sc.tl.paga(a, groups="dcta_leiden", neighbors_key="dcta")
    paga_conn = a.uns["paga"]["connectivities"]
    # Cluster centroids in diffusion-map space (for downstream viz / metrics)
    dm = a.obsm["X_diffmap"][:, :n_dcs]
    labels = a.obs["dcta_leiden"].astype(str).values
    uniq = sorted(np.unique(labels), key=int)
    centroids = np.vstack([dm[labels == k].mean(axis=0) for k in uniq])
    cluster_sizes = np.array([int((labels == k).sum()) for k in uniq])

    uns = a.uns.setdefault("dcta", {})
    uns["paga"] = {
        "connectivities": paga_conn.toarray() if hasattr(paga_conn, "toarray") else np.asarray(paga_conn),
        "cluster_sizes": cluster_sizes,
        "cluster_centroids": centroids,
        "cluster_labels": uniq,
    }
    return a
```

```python
"""DCTA pseudotime stage: multi-root-aware DPT.

Default strategy: HSC marker medoid per connected component, top-K
robustness re-sampling, virtual_hsc_ref hook deferred to Phase 2.
"""
from __future__ import annotations
import numpy as np
import scanpy as sc
import anndata as ad
import yaml
from scipy.sparse.csgraph import connected_components
from dcta import DCTARootError
from dcta import defaults as D


def _load_hsc_markers(path: str) -> list[str]:
    with open(path) as f:
        cfg = yaml.safe_load(f)
    return list(cfg.get("hsc_positive", []))


def _score_hsc(adata: ad.AnnData, markers: list[str]) -> np.ndarray:
    """Per-cell mean expression of HSC markers (from log-normalized X)."""
    present = [g for g in markers if g in adata.var_names]
    if not present:
        raise DCTARootError(
            f"None of the HSC marker genes are present in adata.var_names: {markers}"
        )
    idx = [adata.var_names.get_loc(g) for g in present]
    X = adata.X[:, idx]
    if hasattr(X, "toarray"):
        X = X.toarray()
    return np.asarray(X).mean(axis=1)


def _medoid_cell(coords: np.ndarray) -> int:
    """Index of the cell minimizing summed distance to others (in coords subset)."""
    if coords.shape[0] == 1:
        return 0
    diff = coords[:, None, :] - coords[None, :, :]
    d = np.sqrt((diff ** 2).sum(axis=-1))
    return int(np.argmin(d.sum(axis=1)))


def _connected_components_from_paga(adata: ad.AnnData) -> tuple[np.ndarray, int]:
    """Return (cell_component_id, n_components) using PAGA cluster connectivity.

    Maps Leiden cluster -> component via connected_components on the PAGA
    connectivities matrix, then back to cells via dcta_leiden.
    """
    paga = adata.uns["dcta"]["paga"]
    conn = np.asarray(paga["connectivities"])
    labels = list(paga["cluster_labels"])
    # Convert connectivity to undirected adjacency (>0)
    adj = (conn > 0).astype(int)
    n_comp, cluster_comp = connected_components(adj, directed=False)
    label_to_comp = dict(zip(labels, cluster_comp))
    cell_comp = adata.obs["dcta_leiden"].astype(str).map(label_to_comp).to_numpy()
    return cell_comp.astype(int), int(n_comp)


def _dpt_from_iroot(adata: ad.AnnData, iroot: int) -> np.ndarray:
    """Run scanpy DPT with given iroot, return per-cell pseudotime array."""
    a = adata
    a.uns["iroot"] = int(iroot)
    sc.tl.dpt(a, neighbors_key="dcta")
    return a.obs["dpt_pseudotime"].to_numpy().copy()


def run(
    adata: ad.AnnData,
    *,
    copy: bool = False,
    root_strategy: str = "hsc_marker_medoid",
    hsc_marker_yaml: str | None = None,
    robustness_k: int = D.ROBUSTNESS_K,
    ref_adata: ad.AnnData | None = None,
) -> ad.AnnData:
    """Multi-root-aware DPT.

    Strategies:
      - "hsc_marker_medoid" (default): pick medoid of HSC cluster per
        connected component as iroot. Run DPT once per component. Then
        re-run K times using top-K HSC candidates and store cell-wise std.
      - "virtual_hsc_ref": NotImplementedError (Phase 2).
    """
    if root_strategy == "virtual_hsc_ref":
        raise NotImplementedError("virtual_hsc_ref root strategy is Phase 2")

    if root_strategy != "hsc_marker_medoid":
        raise ValueError(f"Unknown root_strategy: {root_strategy}")

    if hsc_marker_yaml is None:
        raise DCTARootError("hsc_marker_yaml is required for hsc_marker_medoid")

    a = adata.copy() if copy else adata
    markers = _load_hsc_markers(hsc_marker_yaml)
    hsc_score = _score_hsc(a, markers)
    if hsc_score.max() <= 0:
        raise DCTARootError("HSC marker score is non-positive for all cells")

    # Component assignment
    cell_comp, n_comp = _connected_components_from_paga(a)
    a.obs["dcta_component"] = cell_comp

    # Pre-store dpt array (init NaN)
    dpt_combined = np.full(a.n_obs, np.nan, dtype=float)
    root_cells: list[int] = []
    root_clusters: list[str] = []

    # Diffusion-map coords for medoid computation
    n_dcs = int(a.uns["dcta"]["n_dcs"])
    dm = a.obsm["X_diffmap"][:, :n_dcs]
    leiden = a.obs["dcta_leiden"].astype(str).to_numpy()

    for comp_id in range(n_comp):
        mask_comp = (cell_comp == comp_id)
        if mask_comp.sum() < 3:
            continue
        # HSC cluster within this component = leiden cluster with max mean HSC score
        comp_clusters = np.unique(leiden[mask_comp])
        cluster_scores = {c: hsc_score[(leiden == c) & mask_comp].mean()
                          for c in comp_clusters}
        hsc_cluster = max(cluster_scores, key=cluster_scores.get)
        # Medoid of that cluster in diffmap coords
        cluster_mask = (leiden == hsc_cluster) & mask_comp
        cluster_idx = np.where(cluster_mask)[0]
        medoid_local = _medoid_cell(dm[cluster_idx])
        iroot = int(cluster_idx[medoid_local])
        root_cells.append(iroot)
        root_clusters.append(hsc_cluster)

        # Run DPT on full adata with this iroot, then keep only this component's values
        dpt_full = _dpt_from_iroot(a, iroot)
        dpt_combined[mask_comp] = dpt_full[mask_comp]

    a.obs["dcta_dpt"] = dpt_combined

    # Robustness: top-K HSC candidates (global), re-run DPT, compute std
    top_k_idx = np.argsort(-hsc_score)[:robustness_k]
    dpts = np.zeros((robustness_k, a.n_obs), dtype=float)
    for i, root in enumerate(top_k_idx):
        dpts[i, :] = _dpt_from_iroot(a, int(root))
    # Use std across replicates, NaN-safe
    dpt_root_std = np.nanstd(dpts, axis=0)

    # Per-cluster mean HSC marker score (used by dag.build for marker-direction)
    leiden = a.obs["dcta_leiden"].astype(str).to_numpy()
    cluster_hsc_score = {
        c: float(hsc_score[leiden == c].mean())
        for c in np.unique(leiden)
    }

    uns = a.uns.setdefault("dcta", {})
    uns["root_strategy"] = root_strategy
    uns["root_cells"] = [int(x) for x in root_cells]
    uns["root_cluster"] = root_clusters[0] if root_clusters else None
    uns["component_count"] = int(n_comp)
    uns["dpt_root_std"] = dpt_root_std
    uns["cluster_hsc_score"] = cluster_hsc_score
    # Phase 1: maturation = inverse of HSC score (no mature-marker YAML yet)
    uns["cluster_maturation_score"] = {c: -v for c, v in cluster_hsc_score.items()}
    return a
```

```python
"""DCTA G_data construction.

For each candidate PAGA edge above `edge_threshold`, compute TWO directions:
  - "dpt":    low cluster-mean DPT -> high cluster-mean DPT
  - "marker": high cluster HSC-marker score -> low (immature -> mature)

If a label column with seniority mapping is available, score each method's
agreement with biology-derived expected direction and pick the winner. Else
default to "marker". Build G_data from the chosen method's directions, then
FAS-remove cycles.
"""
from __future__ import annotations
import numpy as np
import networkx as nx
import anndata as ad
import yaml
from dcta import DCTAGraphError
from dcta import defaults as D


def _greedy_fas_remove(G: nx.DiGraph) -> list[tuple[str, str, dict]]:
    """Greedy feedback arc set: while G has a cycle, remove the weakest edge
    on a found cycle. Returns the list of removed edges with their data.
    Modifies G in place.
    """
    removed: list[tuple[str, str, dict]] = []
    safety = G.number_of_edges() + 10
    for _ in range(safety):
        try:
            cyc = nx.find_cycle(G, orientation="original")
        except nx.NetworkXNoCycle:
            return removed
        weakest = min(cyc, key=lambda e: G[e[0]][e[1]].get("weight", 0.0))
        u, v = weakest[0], weakest[1]
        data = dict(G[u][v])
        removed.append((u, v, data))
        G.remove_edge(u, v)
    raise DCTAGraphError("FAS greedy did not converge")


def _load_seniority(path: str | None) -> dict[str, int]:
    if path is None:
        return {}
    with open(path) as f:
        cfg = yaml.safe_load(f)
    return dict(cfg.get("seniority", {}))


def _cluster_dominant_label(adata: ad.AnnData, label_column: str | None) -> dict[str, str]:
    """Return {cluster_id: dominant_label} using mode of label_column per cluster.

    Returns empty dict if column missing.
    """
    if label_column is None or label_column not in adata.obs.columns:
        return {}
    leiden = adata.obs["dcta_leiden"].astype(str).to_numpy()
    labels = adata.obs[label_column].astype(str).to_numpy()
    out: dict[str, str] = {}
    for c in np.unique(leiden):
        mask = leiden == c
        if not mask.any():
            continue
        vals, counts = np.unique(labels[mask], return_counts=True)
        out[str(c)] = str(vals[int(np.argmax(counts))])
    return out


def _direction_dpt(cluster_mean_dpt: dict[str, float],
                   ci: str, cj: str) -> tuple[str, str] | None:
    """Return (source, target) for DPT direction, or None if undecidable."""
    di, dj = cluster_mean_dpt.get(ci), cluster_mean_dpt.get(cj)
    if di is None or dj is None or np.isnan(di) or np.isnan(dj):
        return None
    return (ci, cj) if di <= dj else (cj, ci)


def _direction_marker(cluster_hsc_score: dict[str, float],
                      ci: str, cj: str) -> tuple[str, str] | None:
    """Return (source, target) for marker direction.

    Higher HSC score = more immature = source.
    """
    hi, hj = cluster_hsc_score.get(ci), cluster_hsc_score.get(cj)
    if hi is None or hj is None or np.isnan(hi) or np.isnan(hj):
        return None
    return (ci, cj) if hi >= hj else (cj, ci)


def _score_method(direction_map: dict[tuple[str, str], tuple[str, str]],
                  cluster_seniority: dict[str, int]) -> float | None:
    """Fraction of edges where method's direction agrees with seniority.

    Only edges whose BOTH endpoints have a seniority and seniorities differ
    contribute. Returns None if no scoreable edges.
    """
    correct, total = 0, 0
    for (ci, cj), (src, dst) in direction_map.items():
        si, sj = cluster_seniority.get(ci), cluster_seniority.get(dst if src == ci else src)
        # Re-fetch cleanly
        sa = cluster_seniority.get(src)
        sb = cluster_seniority.get(dst)
        if sa is None or sb is None or sa == sb:
            continue
        total += 1
        # Expected: source has lower seniority than target
        if sa < sb:
            correct += 1
    return None if total == 0 else correct / total


def build(
    adata: ad.AnnData,
    *,
    copy: bool = False,
    edge_threshold: float = D.EDGE_THRESHOLD,
    fas_method: str = D.FAS_METHOD,
    transitive_reduction: bool = D.TRANSITIVE_REDUCTION,
    direction_strategy: str = "auto_select",
    seniority_yaml: str | None = None,
    label_column: str | None = "cell_type",
) -> ad.AnnData:
    """Build G_data from PAGA connectivities with dual-direction selection.

    Direction strategy:
      - "auto_select" (default): compute both `dpt` and `marker` directions,
        score each against seniority labels if available, choose higher.
        Falls back to "marker" when labels missing.
      - "marker": force marker direction.
      - "dpt": force DPT direction.
    """
    a = adata.copy() if copy else adata
    if fas_method != "greedy":
        raise NotImplementedError(f"fas_method={fas_method!r} not implemented")
    if direction_strategy not in ("auto_select", "marker", "dpt"):
        raise ValueError(f"direction_strategy={direction_strategy!r} not in "
                         "('auto_select', 'marker', 'dpt')")

    paga = a.uns["dcta"]["paga"]
    conn = np.asarray(paga["connectivities"])
    labels = [str(l) for l in paga["cluster_labels"]]
    dpt = a.obs["dcta_dpt"].to_numpy()
    leiden = a.obs["dcta_leiden"].astype(str).to_numpy()
    cluster_mean_dpt = {c: float(np.nanmean(dpt[leiden == c])) for c in labels}
    cluster_hsc_score = dict(a.uns["dcta"]["cluster_hsc_score"])

    # Compute both direction methods per candidate edge
    candidate_pairs: list[tuple[str, str, float]] = []
    n = len(labels)
    for i in range(n):
        for j in range(i + 1, n):
            w = float(conn[i, j])
            if w < edge_threshold:
                continue
            candidate_pairs.append((labels[i], labels[j], w))

    dpt_dirs: dict[tuple[str, str], tuple[str, str]] = {}
    marker_dirs: dict[tuple[str, str], tuple[str, str]] = {}
    for ci, cj, _w in candidate_pairs:
        d = _direction_dpt(cluster_mean_dpt, ci, cj)
        m = _direction_marker(cluster_hsc_score, ci, cj)
        if d is not None:
            dpt_dirs[(ci, cj)] = d
        if m is not None:
            marker_dirs[(ci, cj)] = m

    # Per-pair agreement
    pairs_in_both = set(dpt_dirs) & set(marker_dirs)
    agreement = sum(1 for k in pairs_in_both if dpt_dirs[k] == marker_dirs[k])
    agreement_rate = (agreement / len(pairs_in_both)) if pairs_in_both else None

    # Score methods against seniority if labels available
    seniority_map = _load_seniority(seniority_yaml)
    cluster_labels_map = _cluster_dominant_label(a, label_column)
    cluster_seniority = {
        c: seniority_map.get(lbl) for c, lbl in cluster_labels_map.items()
        if seniority_map.get(lbl) is not None
    }
    if cluster_seniority:
        acc_dpt = _score_method(dpt_dirs, cluster_seniority)
        acc_marker = _score_method(marker_dirs, cluster_seniority)
        accuracy = {"dpt": acc_dpt, "marker": acc_marker}
    else:
        accuracy = {}

    # Choose method
    if direction_strategy in ("marker", "dpt"):
        chosen = direction_strategy
    elif accuracy and accuracy.get("dpt") is not None and accuracy.get("marker") is not None:
        chosen = "marker" if accuracy["marker"] >= accuracy["dpt"] else "dpt"
    else:
        chosen = "marker"  # default: biology-grounded

    chosen_dirs = marker_dirs if chosen == "marker" else dpt_dirs

    # Build G with chosen directions
    G = nx.DiGraph()
    for c in labels:
        G.add_node(c, mean_dpt=cluster_mean_dpt.get(c, float("nan")),
                   hsc_score=cluster_hsc_score.get(c, float("nan")),
                   seniority=cluster_seniority.get(c))
    pair_weight = {(ci, cj): w for ci, cj, w in candidate_pairs}
    for (ci, cj), (src, dst) in chosen_dirs.items():
        w = pair_weight[(ci, cj)]
        G.add_edge(src, dst, weight=w)

    removed = _greedy_fas_remove(G)

    if transitive_reduction:
        attrs_backup = {n_: dict(G.nodes[n_]) for n_ in G.nodes}
        G = nx.transitive_reduction(G)
        for n_, data in attrs_backup.items():
            if n_ in G.nodes:
                G.nodes[n_].update(data)

    uns = a.uns.setdefault("dcta", {})
    uns["G_data"] = G
    uns["cycle_removal_log"] = [
        {"u": u, "v": v, "weight": d.get("weight", 0.0)} for u, v, d in removed
    ]
    uns["direction_methods"] = {
        "dpt": {f"{k[0]}|{k[1]}": list(v) for k, v in dpt_dirs.items()},
        "marker": {f"{k[0]}|{k[1]}": list(v) for k, v in marker_dirs.items()},
        "agreement_rate": agreement_rate,
        "accuracy": accuracy,
        "n_pairs_compared": len(pairs_in_both),
    }
    uns["direction_method_chosen"] = chosen
    uns["biological_seniority_map"] = cluster_seniority
    return a
```

def paga_vs_dag_overlay(adata: ad.AnnData, path: str) -> None:
    """Plot PAGA (undirected, grey) with G_data directed edges overlaid.

    Edge color encodes method-agreement between marker and DPT directions:
      green solid  = both methods agree
      orange dashed = methods disagree (chosen-method direction shown)
      red dotted  = edge removed by FAS cycle removal (drawn from
                    cycle_removal_log)

    Per spec Sec 11.4.
    """
    paga = adata.uns["dcta"]["paga"]
    conn = np.asarray(paga["connectivities"])
    labels = [str(l) for l in paga["cluster_labels"]]
    G_pa = nx.Graph()
    for c in labels:
        G_pa.add_node(c)
    for i, ci in enumerate(labels):
        for j in range(i + 1, len(labels)):
            if conn[i, j] > 0:
                G_pa.add_edge(ci, labels[j], weight=float(conn[i, j]))

    # Layout from PAGA edge weights
    pos = nx.spring_layout(G_pa, seed=0, weight="weight")

    G_dag: nx.DiGraph = adata.uns["dcta"]["G_data"]
    methods = adata.uns["dcta"].get("direction_methods", {})
    dpt_dirs_raw = methods.get("dpt", {}) or {}
    marker_dirs_raw = methods.get("marker", {}) or {}
    def _key(s):
        return tuple(s.split("|", 1)) if "|" in s else (s, "")
    dpt_dirs = {_key(k): tuple(v) for k, v in dpt_dirs_raw.items()}
    marker_dirs = {_key(k): tuple(v) for k, v in marker_dirs_raw.items()}

    fig, ax = plt.subplots(figsize=(8, 6))
    nx.draw_networkx_edges(G_pa, pos, ax=ax, edge_color="lightgrey", width=1.5)
    nx.draw_networkx_nodes(G_dag, pos, ax=ax, node_color="lightblue",
                           node_size=600, edgecolors="black")
    nx.draw_networkx_labels(G_dag, pos, ax=ax, font_size=9)

    # Categorize each retained edge
    green_edges, orange_edges = [], []
    for u, v in G_dag.edges:
        # Determine pair-key (ci, cj) by sorted PAGA order (labels list)
        key = (u, v) if labels.index(u) < labels.index(v) else (v, u)
        d_dir = dpt_dirs.get(key)
        m_dir = marker_dirs.get(key)
        if d_dir == m_dir and d_dir is not None:
            green_edges.append((u, v))
        else:
            orange_edges.append((u, v))

    if green_edges:
        nx.draw_networkx_edges(G_dag, pos, ax=ax, edgelist=green_edges,
                               edge_color="green", arrows=True, arrowsize=15,
                               width=2.0, style="solid")
    if orange_edges:
        nx.draw_networkx_edges(G_dag, pos, ax=ax, edgelist=orange_edges,
                               edge_color="darkorange", arrows=True,
                               arrowsize=15, width=2.0, style="dashed")
    # Removed edges (FAS) drawn as red dotted from log
    removed = adata.uns["dcta"].get("cycle_removal_log", [])
    if removed:
        red_pairs = [(r["u"], r["v"]) for r in removed]
        nx.draw_networkx_edges(G_dag, pos, ax=ax, edgelist=red_pairs,
                               edge_color="red", arrows=True, arrowsize=12,
                               width=1.5, style="dotted")

    chosen = adata.uns["dcta"].get("direction_method_chosen", "?")
    ax.set_title(f"PAGA (grey) ↔ G_data overlay — chosen direction: {chosen}\n"
                 f"green=methods agree, orange=disagree, red=removed by FAS")
    ax.set_axis_off()
    _save(fig, path)

Runs the full pipeline:
  load -> preprocessing -> diffusion -> connectivity -> pseudotime -> dag -> viz

Usage:

import sys
import json

def parse_args(argv=None):
    p = argparse.ArgumentParser(description="DCTA Phase 1 pipeline")
    p.add_argument("--h5ad", required=True, help="input .h5ad file")
    p.add_argument("--outdir", default="results/phase1_hay", help="output directory")
    p.add_argument("--hsc-yaml", default="data/priors/hsc_markers.yaml",
                   help="YAML file with HSC marker genes")
    p.add_argument("--n-comps-pca", type=int, default=D.N_COMPS_PCA)
    p.add_argument("--n-comps-diffmap", type=int, default=D.N_COMPS_DIFFMAP)
    p.add_argument("--knn-diffmap-k", type=int, default=D.KNN_DIFFMAP_K)
    p.add_argument("--resolution", type=float, default=D.LEIDEN_RESOLUTION)
    p.add_argument("--edge-threshold", type=float, default=D.EDGE_THRESHOLD)
    p.add_argument("--robustness-k", type=int, default=D.ROBUSTNESS_K)
    p.add_argument("--regress-cell-cycle", action="store_true", default=True)
    p.add_argument("--no-regress-cell-cycle", action="store_false",
                   dest="regress_cell_cycle")
    p.add_argument("--transitive-reduction", action="store_true", default=False)
    p.add_argument("--seniority-yaml", default="data/priors/lineage_seniority.yaml",
                   help="YAML mapping cell-type labels to seniority rank (for direction-accuracy)")
    p.add_argument("--label-column", default="cell_type",
                   help="adata.obs column used to look up seniority")
    p.add_argument("--direction-strategy", default="auto_select",
                   choices=["auto_select", "marker", "dpt"])
    return p.parse_args(argv)


def main(argv=None):
    args = parse_args(argv)
    os.makedirs(args.outdir, exist_ok=True)

    print(f"[DCTA] loading {args.h5ad}", flush=True)
    adata = dcta.io.load_h5ad(args.h5ad)
    print(f"[DCTA] adata: {adata.n_obs} cells x {adata.n_vars} genes", flush=True)

    print("[DCTA] preprocessing", flush=True)
    adata = dcta.preprocessing.run(adata, regress_cell_cycle=args.regress_cell_cycle)
    print(f"[DCTA] after QC: {adata.n_obs} cells x {adata.n_vars} genes", flush=True)

    print("[DCTA] diffusion", flush=True)
    adata = dcta.diffusion.run(adata, n_comps_pca=args.n_comps_pca,
                               n_comps=args.n_comps_diffmap)
    print(f"[DCTA] n_dcs selected: {adata.uns['dcta']['n_dcs']}", flush=True)

    print("[DCTA] connectivity", flush=True)
    adata = dcta.connectivity.run(adata, knn_k=args.knn_diffmap_k,
                                  resolution=args.resolution)
    n_clust = adata.obs["dcta_leiden"].nunique()
    print(f"[DCTA] leiden clusters: {n_clust}", flush=True)

    print("[DCTA] pseudotime", flush=True)
    adata = dcta.pseudotime.run(adata, hsc_marker_yaml=args.hsc_yaml,
                                robustness_k=args.robustness_k)

          f"components: {adata.uns['dcta']['component_count']}", flush=True)

    print("[DCTA] dag", flush=True)
    adata = dcta.dag.build(adata, edge_threshold=args.edge_threshold,
                           transitive_reduction=args.transitive_reduction,
                           direction_strategy=args.direction_strategy,
                           seniority_yaml=args.seniority_yaml,
                           label_column=args.label_column)
    G = adata.uns["dcta"]["G_data"]
    chosen = adata.uns["dcta"]["direction_method_chosen"]
    acc = adata.uns["dcta"]["direction_methods"].get("accuracy", {})
    print(f"[DCTA] G_data: {G.number_of_nodes()} nodes, {G.number_of_edges()} edges, "

    adata.write_h5ad(out_h5ad)
    print(f"[DCTA] writing {out_gml}", flush=True)
    # nx.write_gml requires str-keyed attrs and no None values

    for n_, data in G.nodes(data=True):
        G_clean.add_node(str(n_), **{k: float(v) if v is not None else 0.0
                                     for k, v in data.items()})
    for u, v, data in G.edges(data=True):
        G_clean.add_edge(str(u), str(v), **{k: float(val) for k, val in data.items()})
    nx.write_gml(G_clean, out_gml)

    print("[DCTA] exporting figures", flush=True)
    dcta.viz.export_all(adata, outdir=args.outdir)

    methods = adata.uns["dcta"].get("direction_methods", {})
    summary = {
        "n_cells_after_qc": int(adata.n_obs),
        "n_genes_after_qc": int(adata.n_vars),
        "n_dcs": int(adata.uns["dcta"]["n_dcs"]),
        "n_clusters": int(n_clust),
        "n_components": int(adata.uns["dcta"]["component_count"]),
        "root_cluster": adata.uns["dcta"]["root_cluster"],
        "n_edges_initial": int(G.number_of_edges() + len(adata.uns["dcta"]["cycle_removal_log"])),
        "n_edges_after_fas": int(G.number_of_edges()),
        "n_edges_removed": int(len(adata.uns["dcta"]["cycle_removal_log"])),
        "direction": {
            "method_chosen": chosen,

"""End-to-end smoke test on a 1000-cell stratified subset of the Hay BM dataset.

Marked slow — runs only when explicitly selected.
"""
import os
import pytest
import numpy as np
import anndata as ad
import networkx as nx
from pathlib import Path

import dcta

HAY_PATH = "/Users/edemhw/Honglab/CS/04_Claude/10_bm_autoannotation/test_datasets/cxg/hay_hca_balanced_bm.h5ad"
HSC_YAML = str(Path(__file__).resolve().parent.parent / "data" / "priors" / "hsc_markers.yaml")


pytestmark = pytest.mark.slow


@pytest.fixture(scope="module")
def hay_subset():
    if not os.path.exists(HAY_PATH):
        pytest.skip(f"Hay dataset not found at {HAY_PATH}")
    a = ad.read_h5ad(HAY_PATH)
    # Stratified subset: up to 1000 cells, balanced across cell_type if column exists
    rng = np.random.default_rng(0)
    target = 1000
    if "cell_type" in a.obs.columns:
        groups = a.obs.groupby("cell_type").indices
        per_group = max(1, target // max(1, len(groups)))
        picks = []
        for _, idx in groups.items():
            n = min(per_group, len(idx))
            picks.extend(rng.choice(idx, size=n, replace=False).tolist())
        picks = picks[:target]
    else:
        picks = rng.choice(a.n_obs, size=min(target, a.n_obs), replace=False).tolist()
    sub = a[picks, :].copy()
    return sub


def test_hay_phase1_end_to_end(hay_subset, tmp_path):
    a = hay_subset
    a = dcta.preprocessing.run(a, regress_cell_cycle=True)
    a = dcta.diffusion.run(a, n_comps=15)
    a = dcta.connectivity.run(a, resolution=1.0)
    a = dcta.pseudotime.run(a, hsc_marker_yaml=HSC_YAML, robustness_k=3)
    seniority_yaml = str(Path(__file__).resolve().parent.parent / "data" / "priors" / "lineage_seniority.yaml")
    a = dcta.dag.build(a, edge_threshold=0.05, seniority_yaml=seniority_yaml,
                       label_column="cell_type")

    G = a.uns["dcta"]["G_data"]
    assert isinstance(G, nx.DiGraph)
    assert nx.is_directed_acyclic_graph(G)
    assert G.number_of_nodes() >= 3, f"only {G.number_of_nodes()} clusters survived QC"

    # Root cluster reachable to at least one leaf
    root = a.uns["dcta"]["root_cluster"]
    assert root in G.nodes
    leaves = [n for n in G.nodes if G.out_degree(n) == 0]
    reach = [n for n in leaves if nx.has_path(G, root, n)]
    assert len(reach) >= 1, "root cluster cannot reach any leaf"

    # viz export should succeed (6 figures incl. paga_vs_dag overlay)
    dcta.viz.export_all(a, outdir=str(tmp_path))
    for f in ["diffmap_eigvals.png", "paga_connectivities.png",
              "dpt_distribution.png", "dpt_root_std_hist.png",
              "G_data_diagram.png", "paga_vs_dag_overlay.png"]:
        assert (tmp_path / f).exists()
    # Direction-method fields populated
    assert a.uns["dcta"]["direction_method_chosen"] in ("marker", "dpt")
